static volatile bool s_uiSelfTestExitArmed = false;
static volatile uint32_t s_uiSelfTestExitDeadline = 0;

static void uiSelfTestExitTask(void*) {
  for (;;) {
    if (s_uiSelfTestExitArmed &&
        (int32_t)(millis() - s_uiSelfTestExitDeadline) >= 0) {
      feature_exit_requested = true;
      s_uiSelfTestExitArmed = false;
    }
    vTaskDelay(pdMS_TO_TICKS(20));
  }
}

static void uiSelfTestArmExit(uint32_t ms) {
  feature_exit_requested = false;
  s_uiSelfTestExitDeadline = millis() + ms;
  s_uiSelfTestExitArmed = true;
}

static void uiSelfTestDisarmExit() {
  s_uiSelfTestExitArmed = false;
}

static void uiSelfTestResetToMenu() {
  clearInjectedTouchForTest();
  uiSelfTestDisarmExit();
  feature_exit_requested = false;
  feature_active = false;
  in_sub_menu = false;
  is_main_menu = true;
  other_layer = OTHER_LAYER_HOME;
  setTouchButtonInputEnabled(false);
  setTouchNavLabels(nullptr, nullptr, nullptr, nullptr, nullptr);
  resetTouchNavHeldState();
  menu_initialized = false;
  submenu_initialized = false;
  other_menu_grid_initialized = false;
  displayMenu();
  delay(80);
}

static void uiSelfTestTapMain(int mainIdx) {
  current_menu_index = mainIdx;
  displayMenu();
  const int column = mainIdx / 4;
  const int row = mainIdx % 4;
  const int x = menuTileX(column) + MENU_TILE_W / 2;
  const int y = menuTileY(row, MAIN_MENU_ROWS) + MENU_TILE_H / 2;
  ESP_LOGI(UI_TEST_TAG, "TAP main[%d]=%s at %d,%d", mainIdx, menu_items[mainIdx], x, y);
  injectTouchScreenXYForTest(x, y, 1);
  delay(60);
  clearInjectedTouchForTest();
}

static void uiSelfTestTapSubmenu(const char* groupName, int subIdx) {
  current_submenu_index = subIdx;
  submenu_initialized = false;
  displaySubmenu();
  const char* label =
      (active_submenu_items && subIdx >= 0 && subIdx < active_submenu_size)
          ? active_submenu_items[subIdx]
          : "?";
  int yPos = 30 + subIdx * 30;
  if (subIdx == active_submenu_size - 1) {
    yPos += 10;
  }
  ESP_LOGI(UI_TEST_TAG, "TAP %s[%d]=%s at %d,%d", groupName, subIdx, label,
           60, yPos + 12);
  injectTouchScreenXYForTest(60, yPos + 12, 1);
  delay(60);
  clearInjectedTouchForTest();
}

static void uiSelfTestOpenTopSubmenu(int mainIdx) {
  uiSelfTestTapMain(mainIdx);
  updateActiveSubmenu();
  current_submenu_index = 0;
  in_sub_menu = true;
  is_main_menu = false;
  submenu_initialized = false;
  displaySubmenu();
}

static void uiSelfTestOpenOtherLayer(uint8_t layer, int homeIdx, const char* name) {
  uiSelfTestTapMain(2);
  other_layer = OTHER_LAYER_HOME;
  current_menu_index = 2;
  current_submenu_index = homeIdx;
  updateActiveSubmenu();
  in_sub_menu = true;
  submenu_initialized = false;
  displaySubmenu();
  const int gridRows = (other_NUM_SUBMENU_ITEMS + OTHER_GRID_COLS - 1) / OTHER_GRID_COLS;
  const int column = homeIdx % OTHER_GRID_COLS;
  const int row = homeIdx / OTHER_GRID_COLS;
  ESP_LOGI(UI_TEST_TAG, "TAP More/%s at %d,%d", name,
           menuTileX(column) + MENU_TILE_W / 2,
           menuTileY(row, gridRows) + MENU_TILE_H / 2);
  delay(60);
  other_layer = layer;
  current_submenu_index = 0;
  updateActiveSubmenu();
  submenu_initialized = false;
  displaySubmenu();
}

static void uiSelfTestRunLoopFeature(const char* path, int mainIdx, int subIdx,
                                     void (*setupFn)(), void (*loopFn)(),
                                     void (*exitFn)() = nullptr,
                                     uint32_t runMs = 1400) {
  ESP_LOGI(UI_TEST_TAG, "START %s", path);
  uiSelfTestOpenTopSubmenu(mainIdx);
  uiSelfTestTapSubmenu(path, subIdx);
  current_submenu_index = subIdx;
  in_sub_menu = true;
  feature_active = true;
  feature_exit_requested = false;
  setTouchButtonInputEnabled(true);
  setupFn();
  uiSelfTestArmExit(runMs);
  const uint32_t start = millis();
  while (!feature_exit_requested && (uint32_t)(millis() - start) < runMs + 600u) {
    current_submenu_index = subIdx;
    in_sub_menu = true;
    feature_active = true;
    loopFn();
    delay(3);
  }
  uiSelfTestDisarmExit();
  feature_exit_requested = true;
  if (exitFn) {
    exitFn();
  }
  ESP_LOGI(UI_TEST_TAG, "END %s elapsed=%lu", path, (unsigned long)(millis() - start));
  uiSelfTestResetToMenu();
}
