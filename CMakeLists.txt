cmake_minimum_required(VERSION 3.16)
set(DIV_BOARD "v2" CACHE STRING "ESP32-DIV board: v2, v1 or cyd")
set_property(CACHE DIV_BOARD PROPERTY STRINGS v2 v1 cyd)
if(NOT DIV_BOARD MATCHES "^(v2|v1|cyd)$")
    message(FATAL_ERROR "DIV_BOARD must be v2, v1 or cyd")
endif()
if(DIV_BOARD STREQUAL "v2")
    set(DIV_CHIP esp32s3)
else()
    set(DIV_CHIP esp32)
endif()
set(IDF_TARGET ${DIV_CHIP} CACHE STRING "ESP-IDF target")
if(NOT IDF_TARGET STREQUAL DIV_CHIP)
    message(FATAL_ERROR "${DIV_BOARD} requires IDF_TARGET=${DIV_CHIP}; use a separate build directory")
endif()
set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig" CACHE FILEPATH "Generated board configuration")
set(SDKCONFIG_DEFAULTS "sdkconfig.defaults;sdkconfig.defaults.${DIV_BOARD}")
set(PROJECT_VER "2.0.0")
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR NOT IDF_VERSION_MINOR EQUAL 5)
    message(FATAL_ERROR "This project requires ESP-IDF v5.5.x (validated with v5.5.5). Activate the matching ESP-IDF environment.")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(esp32_div_${DIV_BOARD})

string(TOUPPER "${DIV_BOARD}" DIV_BOARD_UPPER)
if(NOT CONFIG_ESP32_DIV_BOARD_${DIV_BOARD_UPPER})
    message(FATAL_ERROR "sdkconfig board does not match DIV_BOARD=${DIV_BOARD}; use a fresh build directory")
endif()
